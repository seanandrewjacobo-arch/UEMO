<?php
// send.php
use PHPMailer\PHPMailer\PHPMailer;
use PHPMailer\PHPMailer\Exception;

require 'PHPMailer/src/Exception.php';
require 'PHPMailer/src/PHPMailer.php';
require 'PHPMailer/src/SMTP.php';

// ------- CONFIG (update) -------
$clubEmail = 'yourclub@gmail.com';
$smtpUser   = 'yourclub@gmail.com';
$smtpPass   = 'YOUR_APP_PASSWORD'; // use Gmail App Password or SMTP credential
// ---------------------------------

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    header("Location: index.html");
    exit;
}

function s($k) {
    return isset($_POST[$k]) ? htmlspecialchars(trim($_POST[$k])) : '';
}

$fullname = s('fullname');
$studentid = s('studentid');
$email = s('email');
$course = s('course');
$year = s('year');
$phone = s('phone');
$availability = s('availability');

$interests = [];
if (!empty($_POST['interest']) && is_array($_POST['interest'])) {
    $interests = array_map('htmlspecialchars', $_POST['interest']);
}
$interestList = $interests ? implode(', ', $interests) : 'None';

$body = "
  <div style='font-family:Arial, Helvetica, sans-serif; color:#0f1724;'>
    <h2 style='color:#0D3B66;'>New UEMO Registration</h2>
    <table style='width:100%; border-collapse:collapse;'>
      <tr><td style='padding:6px; font-weight:700;'>Full name</td><td style='padding:6px;'>{$fullname}</td></tr>
      <tr><td style='padding:6px; font-weight:700;'>Student ID</td><td style='padding:6px;'>{$studentid}</td></tr>
      <tr><td style='padding:6px; font-weight:700;'>Email</td><td style='padding:6px;'>{$email}</td></tr>
      <tr><td style='padding:6px; font-weight:700;'>Course</td><td style='padding:6px;'>{$course}</td></tr>
      <tr><td style='padding:6px; font-weight:700;'>Year Level</td><td style='padding:6px;'>{$year}</td></tr>
      <tr><td style='padding:6px; font-weight:700;'>Contact Number</td><td style='padding:6px;'>{$phone}</td></tr>
      <tr><td style='padding:6px; font-weight:700;'>Areas of interest</td><td style='padding:6px;'>{$interestList}</td></tr>
      <tr><td style='padding:6px; font-weight:700;'>Availability</td><td style='padding:6px;'>{$availability}</td></tr>
    </table>
    <p style='margin-top:12px; color:#6b7280;'>Submitted: " . date('Y-m-d H:i:s') . "</p>
  </div>
";

$mail = new PHPMailer(true);

try {
    $mail->isSMTP();
    $mail->Host       = 'smtp.gmail.com';
    $mail->SMTPAuth   = true;
    $mail->Username   = $smtpUser;
    $mail->Password   = $smtpPass;
    $mail->SMTPSecure = PHPMailer::ENCRYPTION_STARTTLS;
    $mail->Port       = 587;

    $mail->setFrom($smtpUser, 'UEMO Registration');
    $mail->addAddress($clubEmail);
    $mail->addReplyTo($email, $fullname);

    $mail->isHTML(true);
    $mail->Subject = "UEMO Registration: {$fullname} ({$studentid})";
    $mail->Body    = $body;
    $mail->AltBody = strip_tags($body);

    $mail->send();

    // show a simple confirmed page
    echo "<!doctype html><html><head><meta charset='utf-8'><meta name='viewport' content='width=device-width,initial-scale=1'>";
    echo "<title>Registration Received</title>";
    echo "<style>body{font-family:Inter, Arial, sans-serif;background:#f6f8fa;color:#0f1724;padding:30px} .card{max-width:720px;margin:40px auto;background:#fff;padding:22px;border-radius:10px;box-shadow:0 8px 24px rgba(16,24,40,.06)} a{color:#0D3B66;text-decoration:none;font-weight:700}</style>";
    echo "</head><body><div class='card'><h2 style='font-family:Merriweather, serif;color:#0D3B66'>Registration Received</h2><p>Thank you, <strong>" . htmlspecialchars($fullname) . "</strong>. Your registration has been received. The Student Affairs office will review and contact you at <strong>" . htmlspecialchars($email) . "</strong> within 24–48 business hours.</p><p><a href='index.html'>&larr; Back to site</a></p></div></body></html>";
    exit;
} catch (Exception $e) {
    error_log("Mailer Error: {$mail->ErrorInfo}");
    echo "<!doctype html><html><head><meta charset='utf-8'><meta name='viewport' content='width=device-width,initial-scale=1'><title>Error</title>
    <style>body{font-family:Inter, Arial, sans-serif;padding:30px;color:#0f1724;background:#f6f8fa} .err{max-width:720px;margin:40px auto;background:#fff;border-radius:10px;padding:20px;box-shadow:0 6px 20px rgba(16,24,40,.06)}</style></head><body><div class='err'><h2 style='font-family:Merriweather, serif;color:#a73b3b'>Submission Error</h2><p>There was a problem sending your registration. Please contact the Student Affairs office or try again later.</p><p><a href='index.html'>&larr; Back to site</a></p></div></body></html>";
    exit;
}
